cmake_minimum_required(VERSION 3.16)
project(constrained_xpbd_round2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
 set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(XPBD_ROOT "/home/blacknoir/Projects/constrained-xpbd" CACHE PATH "Constrained XPBD checkout")
set(COMPOSITION_SOURCE "${XPBD_ROOT}/experiments/composition_2026_10_31" CACHE PATH "Round-one experiment source")
find_package(OpenMP REQUIRED)
include_directories("${XPBD_ROOT}/external/eigen" "${COMPOSITION_SOURCE}")
add_executable(adversarial_study adversarial/study.cpp)
add_executable(radial_mode_probe adversarial/probe.cpp)
add_executable(objective_study objectives/run.cpp)
add_executable(objective_near_rest objectives/near_rest.cpp)
add_executable(objective_moving_probe objectives/moving_probe.cpp)
add_executable(preconditioning_study preconditioning/benchmark.cpp)
add_executable(local_work_study local_work/study.cpp)
add_executable(local_work_rollouts local_work/al_rollouts.cpp)
add_executable(local_work_cross local_work/cross_objective.cpp)
add_executable(local_work_scalar_stalls local_work/scalar_stalls.cpp)
foreach(target adversarial_study radial_mode_probe objective_study objective_near_rest objective_moving_probe preconditioning_study local_work_study local_work_rollouts local_work_cross local_work_scalar_stalls)
 target_link_libraries(${target} PRIVATE OpenMP::OpenMP_CXX)
endforeach()
enable_testing()
add_test(NAME adversarial_validation COMMAND adversarial_study validate)
add_test(NAME objective_validation COMMAND objective_study --validate-only)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/objective_validation")
set_tests_properties(objective_validation PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/objective_validation")
add_test(NAME preconditioning_validation COMMAND preconditioning_study --tests)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/preconditioning_validation")
set_tests_properties(preconditioning_validation PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/preconditioning_validation")
add_test(NAME local_work_validation COMMAND local_work_study validate)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/local_work_validation")
set_tests_properties(local_work_validation PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/local_work_validation")
