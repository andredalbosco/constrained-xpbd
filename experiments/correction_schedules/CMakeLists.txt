cmake_minimum_required(VERSION 3.20)
project(xpbd_correction_schedules LANGUAGES CXX)
include(CTest)
function(correction_target target source)
    add_executable(${target} ${source})
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_include_directories(${target} PRIVATE ../gradient_policies ../energy_ordering)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()
correction_target(xpbd-correction-schedules-study study.cpp)
correction_target(xpbd-correction-followups followups.cpp)
correction_target(xpbd-correction-assembly-probe assembly_probe.cpp)
correction_target(xpbd-warm-start-probe warm_start_probe.cpp)
correction_target(xpbd-symplectic-probe symplectic_probe.cpp)
correction_target(xpbd-symplectic-energy-study symplectic_energy_study.cpp)
if(BUILD_TESTING)
    correction_target(xpbd-correction-schedules-tests tests.cpp)
    add_test(NAME correction-schedules COMMAND xpbd-correction-schedules-tests)
    add_test(NAME correction-stable-assembly COMMAND xpbd-correction-assembly-probe
        "${CMAKE_CURRENT_BINARY_DIR}/assembly_probe.csv")
    add_test(NAME incremental-warm-start COMMAND xpbd-warm-start-probe
        "${CMAKE_CURRENT_BINARY_DIR}/warm_start_results")
    add_test(NAME warm-start-symplecticity COMMAND xpbd-symplectic-probe
        "${CMAKE_CURRENT_BINARY_DIR}/symplectic_results")
    add_test(NAME symplectic-energy-correction COMMAND xpbd-symplectic-energy-study --test)
endif()
