cmake_minimum_required(VERSION 3.20)
project(xpbd_research_agenda LANGUAGES CXX)
include(CTest)
function(agenda_target name source)
  add_executable(${name} ${source})
  target_compile_features(${name} PRIVATE cxx_std_17)
  target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(${name} PRIVATE .. ../gradient_policies ../energy_ordering)
endfunction()
agenda_target(xpbd-agenda-study study.cpp)
agenda_target(xpbd-agenda-conditioning conditioning.cpp)
agenda_target(xpbd-agenda-stability stability.cpp)
if(BUILD_TESTING)
  agenda_target(xpbd-agenda-tests tests.cpp)
  add_test(NAME agenda-invariants-and-derivatives COMMAND xpbd-agenda-tests)
endif()
