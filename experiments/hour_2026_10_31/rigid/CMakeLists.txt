cmake_minimum_required(VERSION 3.16)
project(xpbd_rigid_chain_study LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP REQUIRED)
set(COMPOSITION_ROOT "/home/blacknoir/Projects/constrained-xpbd/experiments/composition_2026_10_31" CACHE PATH "Round-one composition headers")
set(ROUND2_ROOT "/home/blacknoir/Projects/constrained-xpbd/experiments/round2_2026_10_31" CACHE PATH "Round-two objective headers")
set(EIGEN_ROOT "/home/blacknoir/Projects/constrained-xpbd/external/eigen" CACHE PATH "Eigen include root")
foreach(pair IN ITEMS "rigid_study:study.cpp" "rigid_probes:probes.cpp" "rigid_modes:modes.cpp" "rigid_modal_transfer:modal_transfer.cpp" "rigid_controller_probe:controller_probe.cpp")
  string(REPLACE ":" ";" parts "${pair}")
  list(GET parts 0 target)
  list(GET parts 1 source)
  add_executable(${target} ${source})
  target_include_directories(${target} PRIVATE "${EIGEN_ROOT}" "${COMPOSITION_ROOT}" "${ROUND2_ROOT}/objectives")
  target_link_libraries(${target} PRIVATE OpenMP::OpenMP_CXX)
endforeach()
enable_testing()
add_test(NAME rigid_validation COMMAND rigid_study validate)
add_test(NAME rigid_derivative_validation COMMAND rigid_probes)
set_tests_properties(rigid_derivative_validation PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
