cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(constrained_xpbd_local_solver LANGUAGES CXX)
  include(CTest)
endif()

set(COMPOSITION_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../composition_2026_10_31" CACHE PATH "Prior composition experiment containing common.hpp and independent.hpp")
if(NOT DEFINED ROUND2_ROOT)
  set(_round2_default "${CMAKE_CURRENT_SOURCE_DIR}/../../round2_2026_10_31")
  foreach(_candidate IN ITEMS
      "${CMAKE_CURRENT_SOURCE_DIR}/../../composition_round2_2026_10_31"
      "${CMAKE_CURRENT_SOURCE_DIR}/../../constrained-xpbd-round2-2026-10-31")
    if(NOT EXISTS "${_round2_default}/local_work/cross_problem.hpp" AND EXISTS "${_candidate}/local_work/cross_problem.hpp")
      set(_round2_default "${_candidate}")
    endif()
  endforeach()
  set(ROUND2_ROOT "${_round2_default}" CACHE PATH "Round2 experiment containing local_work/cross_problem.hpp")
endif()
set(EIGEN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../external/eigen" CACHE PATH "Eigen include root containing Eigen/Dense")
foreach(_required IN ITEMS "${COMPOSITION_ROOT}/independent.hpp" "${ROUND2_ROOT}/local_work/cross_problem.hpp" "${EIGEN_ROOT}/Eigen/Dense")
  if(NOT EXISTS "${_required}")
    message(FATAL_ERROR "Missing ${_required}; set COMPOSITION_ROOT, ROUND2_ROOT, and EIGEN_ROOT to the saved experiment/include directories.")
  endif()
endforeach()
find_package(Threads REQUIRED)

foreach(_program IN ITEMS study branches block_bench long_blocks spectrum rod_control incremental_checks warm_blocks)
  add_executable(hour_local_${_program} "${_program}.cpp")
  target_compile_features(hour_local_${_program} PRIVATE cxx_std_17)
  target_include_directories(hour_local_${_program} PRIVATE "${ROUND2_ROOT}/local_work" "${COMPOSITION_ROOT}" "${EIGEN_ROOT}")
  target_link_libraries(hour_local_${_program} PRIVATE Threads::Threads)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(hour_local_${_program} PRIVATE -O2)
  endif()
endforeach()

if(BUILD_TESTING)
  set(_validation_output "${CMAKE_CURRENT_BINARY_DIR}/validation_results")
  add_test(NAME hour_local_derivatives COMMAND hour_local_study validate "${_validation_output}")
  add_test(NAME hour_local_incremental_difference COMMAND hour_local_incremental_checks "${_validation_output}")
  add_test(NAME hour_local_branches_and_reference COMMAND hour_local_branches "${_validation_output}")
  add_test(NAME hour_local_iteration_spectrum COMMAND hour_local_spectrum "${_validation_output}")
  set_tests_properties(hour_local_derivatives hour_local_incremental_difference hour_local_branches_and_reference hour_local_iteration_spectrum PROPERTIES TIMEOUT 120)
endif()
