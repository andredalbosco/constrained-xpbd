cmake_minimum_required(VERSION 3.16)
project(xpbd_work_reduction LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
 set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(XPBD_ROOT "/home/blacknoir/Projects/constrained-xpbd" CACHE PATH "Main repository")
set(COMPOSITION_ROOT "${XPBD_ROOT}/experiments/composition_2026_10_31" CACHE PATH "First experiment headers")
set(ROUND2_ROOT "${XPBD_ROOT}/experiments/round2_2026_10_31" CACHE PATH "Second experiment package")
set(EIGEN_ROOT "${XPBD_ROOT}/external/eigen" CACHE PATH "Eigen headers")
add_executable(work_reduction_study study.cpp)
add_executable(work_normalization normalization.cpp)
add_executable(work_reduction_larger larger.cpp)
add_executable(work_reduction_nonlinear nonlinear.cpp)
add_executable(work_disconnected disconnected.cpp)
add_executable(work_reduction_rollouts rollouts.cpp)
add_executable(work_reduction_reference reference_audit.cpp)
foreach(target work_reduction_study work_normalization work_reduction_larger work_reduction_nonlinear work_disconnected work_reduction_rollouts work_reduction_reference)
 target_include_directories(${target} PRIVATE "${EIGEN_ROOT}" "${COMPOSITION_ROOT}" "${ROUND2_ROOT}/local_work" "${ROUND2_ROOT}/objectives")
endforeach()
enable_testing()
add_test(NAME work_reduction_derivatives COMMAND work_reduction_study "${CMAKE_CURRENT_BINARY_DIR}/validation" validate)
add_test(NAME work_reduction_analytic_oracle COMMAND work_reduction_larger "${CMAKE_CURRENT_BINARY_DIR}/analytic_validation" validate)
