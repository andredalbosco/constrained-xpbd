cmake_minimum_required(VERSION 3.16)
project(xpbd_longer_rollouts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(XPBD_ROOT "$ENV{HOME}/Projects/constrained-xpbd" CACHE PATH "Repository")
set(HOUR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "Frozen hour package")
set(COMPOSITION_ROOT "${XPBD_ROOT}/experiments/composition_2026_10_31" CACHE PATH "First experiment headers")
set(ROUND2_ROOT "${XPBD_ROOT}/experiments/round2_2026_10_31" CACHE PATH "Second experiment headers")
set(EIGEN_ROOT "${XPBD_ROOT}/external/eigen" CACHE PATH "Eigen headers")
foreach(source rollouts reference_audit reference_fine)
 add_executable(longer_${source} ${source}.cpp)
 target_include_directories(longer_${source} PRIVATE "${EIGEN_ROOT}" "${COMPOSITION_ROOT}" "${ROUND2_ROOT}/local_work" "${HOUR_ROOT}/work_reduction" "${HOUR_ROOT}/local_solver")
endforeach()
