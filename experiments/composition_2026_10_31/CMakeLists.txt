cmake_minimum_required(VERSION 3.20)
project(xpbd_composition LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(XPBD_SOURCE /home/blacknoir/Projects/constrained-xpbd CACHE PATH "Existing constrained-xpbd checkout")
find_package(OpenMP REQUIRED)
add_library(settings INTERFACE)
target_include_directories(settings INTERFACE ${XPBD_SOURCE}/external/eigen ${XPBD_SOURCE}/experiments ${XPBD_SOURCE}/experiments/gradient_policies ${XPBD_SOURCE}/experiments/energy_ordering)
target_compile_options(settings INTERFACE -Wall -Wextra -Wpedantic)
add_executable(study study.cpp)
target_link_libraries(study PRIVATE settings OpenMP::OpenMP_CXX)
add_executable(landscape landscape.cpp)
target_link_libraries(landscape PRIVATE settings)
enable_testing()
foreach(name IN ITEMS independent_validation mechanics_validation audits)
  add_executable(${name} ${name}.cpp)
  target_link_libraries(${name} PRIVATE settings)
endforeach()
add_test(NAME independent_derivatives_and_identities COMMAND independent_validation)
add_test(NAME mechanics_constraints_and_convergence COMMAND mechanics_validation)
add_test(NAME work_and_impulse_identities COMMAND audits ${CMAKE_CURRENT_BINARY_DIR}/audit_validation)
