cmake_minimum_required(VERSION 3.20)
project(xpbd_traditional_cases LANGUAGES CXX)
include(CTest)
function(traditional_target target source)
    add_executable(${target} ${source})
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_include_directories(${target} PRIVATE .. ../gradient_policies ../energy_ordering)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endfunction()
traditional_target(xpbd-traditional-study study.cpp)
if(BUILD_TESTING)
    traditional_target(xpbd-traditional-tests tests.cpp)
    add_test(NAME traditional-baselines-and-schedules COMMAND xpbd-traditional-tests)
    traditional_target(xpbd-relinearized-tests relinearized_tests.cpp)
    add_test(NAME relinearized-EL-LE COMMAND xpbd-relinearized-tests)
endif()
